b) The Data

Source: Abdallah, S. & Barberà Mas, E. (2026). The Happy Planet Index 2026 Data File. Accessed from www.happyplanetindex.org. Covers all countries, 2006-2025, with Life Expectancy, Life Satisfaction, Ecological Footprint, HPI, GDP per capita, and Continent grouping.

# ---- Data setup ----
hpi_raw <- read_excel("Happy-Planet-Index-2006-2025-public-data-set.xlsx",
                       sheet = "All Data", skip = 1)

continent_lookup <- c(
  "1" = "Latin America", "2" = "N America & Oceania", "3" = "Western Europe",
  "4" = "Middle East & N. Africa", "5" = "Africa", "6" = "South Asia",
  "7" = "Eastern Europe & Central Asia", "8" = "East Asia"
)

hpi <- hpi_raw %>%
  rename(GDP_per_capita = `GDP per capita (cleaned) - 2024 used for 2025`,
         Life_Expectancy = `Life Expectancy`,
         Life_Satisfaction = `Life Satisfaction`,
         Ecological_Footprint = `Ecological Footprint`) %>%
  filter(!is.na(Continent), !is.na(Year)) %>%
  mutate(Continent = continent_lookup[as.character(Continent)])

rainbow <- c(
  "Africa" = "#FFADAD", "East Asia" = "#FFC6FF",
  "Eastern Europe & Central Asia" = "#F0E68C", "Latin America" = "#CAFCCF",
  "Middle East & North Africa" = "#9BF6FF", "North America & Oceania" = "#A0C4FF",
  "South Asia" = "#BDB2FF", "Western Europe" = "#FFD6A5"
)

num_vars18 <- c(
  "HPI"                              = "HPI",
  "HPI rank"                         = "HPI rank",
  "Life Expectancy"                  = "Life_Expectancy",
  "Life Satisfaction"                = "Life_Satisfaction",
  "Ecological Footprint"             = "Ecological_Footprint",
  "Population (thousands)"           = "Population",
  "GDP per capita (cleaned)"         = "GDP_per_capita",
  "GDP per capita (raw)"             = "GDP per capita",
  "CoV adjusted ladder"              = "CoV adjusted ladder",
  "Adj Lad x Life Exp"               = "Adj Lad x Life Exp",
  "Extra HPLY (above min)"           = "Extra HPLY (above min)",
  "CoV Adj Footprint"                = "CoV Adj Footprint",
  "HPLY/EF"                          = "HPLY/EF",
  "Footprint category"               = "Footprint category",
  "EF biocapacity for year"          = "EF biocapacity for year"
)

countries18 <- sort(unique(hpi$Country))
default_countries18 <- c("Brazil", "China", "Germany", "India", "Japan",
                         "Mexico", "Nigeria", "United States of America")

# ---- Facet options ----
facet_vars2 <- c("Continent", "Footprint category", "Country")

footprint_lookup <- c(
  "1" = "Within biocapacity",
  "2" = "Up to 2x biocapacity",

server <- function(input, output) {
  thematic::thematic_shiny()
  
  # Shared reactive used by ALL THREE charts - everyone filters off this
  filtered_data <- reactive({
    hpi %>% filter(Year == input$year)
  })

Chart 9: Scatter — Distribution of Two Variables

Solbee Kim

This chart shows the distribution of GDP per capita against Life Satisfaction (or other variable pairs, user-selectable) across all countries for a selected year, colored by continent. It uses a shared reactive() that filters the dataset by year, which also feeds Charts 18 and 3 below — so all three charts always reflect the same year’s data.

  # --- Chart 9: scatter (Solbee) ---
  output$chart9_scatter <- renderPlot({     
    df <- filtered_data()
    ggplot(df, aes(x = .data[[input$xvar9]], y = .data[[input$yvar9]], fill = Continent)) +
      geom_point(shape = 21, color = "grey30", size = 3, stroke = 0.4, alpha = 1) +
      scale_fill_manual(values = rainbow) +
      labs(title = paste("\nDistribution:", input$xvar9, "vs", input$yvar9, "-", input$year, "\n"),
           x = input$xvar9, y = input$yvar9,
           caption = "Source: Happy Planet Index 2006-2025 Data File (happyplanetindex.org)") +
      theme_minimal(base_size = 14)+
      theme(axis.title   = element_text(size = 14),   
            axis.text    = element_text(size = 14),   
            legend.title = element_text(size = 14),   
            legend.text  = element_text(size = 14),
            plot.caption = element_text(hjust = 0,margin = margin(t = 20)),
            plot.caption.position = "plot")   
  })

Chart 18: Bubble - Relationship Among Three Variables

John Vasconcelos

This chart compares Life Expectancy against Life Satisfaction (or other variable pairs, user-selectable) for a selected group of countries in a given year, with bubble size representing a third variable (Population by default, or any of the numeric variables in the dataset). Each bubble is colored by country, and users can add or remove countries through a searchable selector. Like Chart 9, it draws from the shared reactive() that filters the dataset by year, so the bubble chart always reflects the same year selected on the slider.

  # --- Chart 18: bubble (John) ---
  output$chart18_bubble <- renderPlot({
    validate(need(length(input$country18) > 0, "Select at least one country."))
    
    df <- filtered_data() %>%
      filter(Country %in% input$country18,
             !is.na(.data[[input$xvar18]]),
             !is.na(.data[[input$yvar18]]),
             !is.na(.data[[input$sizevar18]]))
    
    validate(need(nrow(df) > 0,
                  "No data for the selected countries and variables in this year."))
    
    lab_x    <- names(num_vars18)[num_vars18 == input$xvar18]
    lab_y    <- names(num_vars18)[num_vars18 == input$yvar18]
    lab_size <- names(num_vars18)[num_vars18 == input$sizevar18]
    
    ggplot(df, aes(x = .data[[input$xvar18]], y = .data[[input$yvar18]],
                   size = .data[[input$sizevar18]], fill = Country)) +
      geom_point(shape = 21, color = "grey30", stroke = 0.4, alpha = 0.6) +
      scale_size(range = c(3, 18), name = lab_size, labels = scales::label_comma()) +
      guides(fill = guide_legend(override.aes = list(size = 4))) +
      labs(title = paste("\nBubble:", lab_x, "vs", lab_y, "(size:", lab_size, ") -", input$year, "\n"),
           x = lab_x, y = lab_y, fill = "Country",
           caption = "Source: Happy Planet Index 2006-2025 Data File (happyplanetindex.org)") +
      theme_minimal(base_size = 14)+
      theme(axis.title   = element_text(size = 14),   
            axis.text    = element_text(size = 14),   
            legend.title = element_text(size = 14),   
            legend.text  = element_text(size = 14),
            plot.caption = element_text(hjust = 0,margin = margin(t = 20)),
            plot.caption.position = "plot") 
  })

Chart 2: Table with Embedded Charts - Comparison Among Many Categories

Evandro Gomes

This chart is a table of small multiples. It suits comparing one variable across many categories. The user picks a Y variable and how to split the panels by Continent, Footprint category, or Country. The panels show the mean of the selected variable for that group in every year from 2006 up to the year chosen on the slider. The panels grow as the slider moves.

# --- Chart 2: Table of charts (Evandro) ---
output$chart2_facets <- renderPlot({
  facet_var <- input$facet2
  df <- hpi %>% filter(Year <= input$year)
  
  if (facet_var == "Country") {
    validate(need(length(input$country2) > 0, "Select at least one country."))
    df <- df %>% filter(Country %in% input$country2)
  }
  
  if (facet_var == "Footprint category") {
    df$Group <- footprint_lookup[as.character(df$`Footprint category`)]
  } else {
    df$Group <- df[[facet_var]]
  }
  
  df_mean <- df %>%
    filter(!is.na(Group)) %>%
    group_by(Group, Year) %>%
    summarise(value = mean(.data[[input$yvar2]], na.rm = TRUE), .groups = "drop") %>%
    filter(!is.nan(value))
  
  validate(need(nrow(df_mean) > 0, "No data for this selection."))
  
  lab_y <- names(num_vars18)[num_vars18 == input$yvar2]
  
  groups2 <- sort(unique(df_mean$Group))
  pal2 <- if (facet_var == "Continent") {
    rainbow
  } else {
    setNames(rep(unname(rainbow), length.out = length(groups2)), groups2)
  }
  
  ggplot(df_mean, aes(x = Year, y = value)) +
    geom_col(aes(fill = Group), color = "grey30", linewidth = 0.4, width = 0.8, show.legend = FALSE) +
    scale_fill_manual(values = pal2) +
    facet_wrap(~ Group) +
    scale_x_continuous(breaks = function(lims) unique(floor(pretty(lims, n = 4)))) +
    scale_y_continuous(labels = scales::label_comma()) +
    labs(title = paste("\nMean", lab_y, "by", facet_var, "-", min(df_mean$Year), "to", input$year, "\n"),
         x = "Year", y = paste("Mean", lab_y),
         caption = "Source: Happy Planet Index 2006-2025 Data File (happyplanetindex.org)") +
    theme_minimal(base_size = 14)+
    theme(strip.text = element_text(size = 12, face = "bold"),
          axis.title   = element_text(size = 14),   
          axis.text    = element_text(size = 14),   
          legend.title = element_text(size = 14),   
          legend.text  = element_text(size = 14),
          plot.caption = element_text(hjust = 0,margin = margin(t = 20)),
          plot.caption.position = "plot") 
})
}
